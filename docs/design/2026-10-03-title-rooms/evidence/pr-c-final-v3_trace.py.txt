"""V3 on the GPU: per main command buffer around each switch from the hall's
frame shape (9 encoders) to the title's (6), its GPU busy, its encoders'
fragment time and the GPU performance state when it started.
usage: v3_trace.py <frames.json> <x-dir> [before] [after]"""
import json, sys, bisect
sys.path.insert(0, 'tools/map_trace')
from xtable import rows
fs = json.load(open(sys.argv[1]))
_, ps = rows(sys.argv[2] + "/gpu-performance-state-intervals.xml")
states = sorted((int(r["start"][1]), int(r["start"][1]) + int(r["duration"][1]), r["gpu-performance-state"][0]) for r in ps)
starts = [s for s, _, _ in states]
def state(t):
    i = bisect.bisect_right(starts, t) - 1
    return states[i][2][:3] if i >= 0 and states[i][1] >= t else "-"
before = int(sys.argv[3]) if len(sys.argv) > 3 else 28
after = int(sys.argv[4]) if len(sys.argv) > 4 else 6
big = [f for f in fs if len(f["encoders"]) >= 5]
t0 = big[0]["start"]
for k in range(1, len(big)):
    if len(big[k - 1]["encoders"]) == 9 and len(big[k]["encoders"]) == 6:
        print("== switch at %.3f s" % ((big[k]["start"] - t0) / 1e9))
        for j in range(max(0, k - before), min(len(big), k + after)):
            f = big[j]
            gap = (f["start"] - big[j - 1]["start"]) / 1e6 if j else 0
            fr = " ".join("%4.1f" % (e["f"] / 1e6) for e in f["encoders"] if e["label"].find("Render") >= 0)
            print("%+3d  start+%5.1f  busy %5.1f  %s  %d enc | F %s" % (j - k, gap, f["busy"] / 1e6, state(f["start"]), len(f["encoders"]), fr))
