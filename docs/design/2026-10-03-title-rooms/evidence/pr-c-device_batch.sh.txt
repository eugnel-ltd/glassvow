#!/bin/zsh
# The #655 PR C iPad 8 batch (docs/design/2026-10-03-title-rooms §11.6). Holds the
# shared iPad's batch lock from the QA install to the last launch (released on any
# exit), builds the QA app (io.fol2.glassvow.qa, never the production bundle) from
# the worktree with the QA-only bench patch applied and reverted at once, and
# launches it once per label with `--rooms-bench` and a per-launch nonce. Before
# every launch the installed QA app must still be the one this batch installed;
# after it, the rows must carry that launch's nonce and the build label; a stale or
# foreign row set is discarded and the launch redone (up to two times). A file copy
# from the QA container that has not answered in 15 minutes ends the batch.
# usage: device_batch.sh <worktree> <out-dir> <build-label>
set -uo pipefail
WT=$1; OUT=$2; BUILD=$3
FLOW=/private/tmp/claude-501/-Users-jamesto-Coding-glassvow/b76db86a-b79e-49a2-b973-bc87e903afff/scratchpad/device13/flow
HERE=${0:A:h}
OWNER=rooms-c
B=io.fol2.glassvow.qa
export DEVELOPER_DIR=/Applications/Xcode.app/Contents/Developer
mkdir -p $OUT
log() { echo "$(date +%T) $*" | tee -a $OUT/batch.log; }
cleanup() {
  python3 $HERE/qa_rooms_patch.py $WT --revert 2>/dev/null
  $FLOW/ipad_lock.sh release $OWNER
  log "lock released"
}
trap cleanup EXIT
udid() { xcrun devicectl list devices --quiet --json-output - 2>/dev/null | python3 -c '
import json,sys
for d in json.load(sys.stdin)["result"]["devices"]:
    h=d.get("hardwareProperties") or {}
    if h.get("productType")=="iPad11,6": print(h.get("udid")); break'; }
app_url() { xcrun devicectl device info apps --device $1 --json-output - 2>/dev/null | python3 -c '
import json,sys
for a in json.load(sys.stdin)["result"]["apps"]:
    if a.get("bundleIdentifier")=="io.fol2.glassvow.qa": print(a.get("url"))'; }

START=$(date +%s)
log "acquiring the iPad batch lock as $OWNER"
$FLOW/ipad_lock.sh acquire $OWNER || exit 1
log "lock held; building $BUILD"
python3 $HERE/qa_rooms_patch.py $WT $BUILD || exit 1
IPAD_LOCK_OWNER=$OWNER $FLOW/qa_export.sh $WT > $OUT/qa_export.out 2>&1
RC=$?
python3 $HERE/qa_rooms_patch.py $WT --revert
[[ $RC -eq 0 ]] || { log "QA export failed"; tail -5 $OUT/qa_export.out | tee -a $OUT/batch.log; exit 1; }
MINE=$(grep -o 'installationURL: [^ ]*' $OUT/qa_export.out | awk '{print $2}' | head -1)
log "installed $MINE"
COPY_STUCK=0
launch() {  # label locale extra-args...
  local label=$1 locale=$2; shift 2
  local f=$OUT/$label.jsonl
  [[ $COPY_STUCK -eq 1 ]] && { log "$label: skipped, file copies hang"; return 1; }
  for attempt in 1 2; do
    local U=$(udid); [[ -n "$U" ]] || { log "no iPad"; return 1; }
    if [[ -n "$MINE" && "$(app_url $U)" != "$MINE" ]]; then log "$label: the QA app is not this batch's build; stopping"; return 1; fi
    local nonce="$label-$(date +%s)"
    rm -f $f
    $FLOW/ipad_lock.sh acquire $OWNER
    xcrun devicectl device process launch --terminate-existing --device $U $B -- --rooms-bench --qa-nonce=$nonce --locale=$locale "$@" > $OUT/launch-$label.log 2>&1
    log "$label: launched ($nonce, attempt $attempt)"
    sleep 90
    local since=$(date +%s)
    for k in $(seq 1 90); do
      timeout 90 xcrun devicectl device copy from --device $U --domain-type appDataContainer --domain-identifier $B \
        --source Documents/bench_rooms.jsonl --destination $f > /dev/null 2>&1
      local rc=$?
      if [[ -f $f ]] && grep -q '"probe":"done"' $f; then break; fi
      if [[ $rc -ne 0 && $(( $(date +%s) - since )) -gt 900 ]]; then
        log "$label: the QA container's file copy has not answered for 15 minutes; stopping the batch"
        COPY_STUCK=1; return 1
      fi
      [[ $rc -eq 0 ]] && since=$(date +%s)
      sleep 10
    done
    if [[ -f $f ]] && grep -q -- "\"nonce\":\"$nonce\"" $f && grep -q '"probe":"done"' $f && grep -q "ipad8-$BUILD" $f; then
      log "$label: $(grep -c . $f) rows, nonce and build match ($(( ($(date +%s) - START) / 60 )) min into the batch)"
      return 0
    fi
    log "$label: attempt $attempt stale or incomplete; discarded"
    mv $f $f.discarded-$attempt 2>/dev/null
  done
  return 1
}
launch en-1 en
launch zh-2 zh-Hant
launch a0-3 en --bench-act=0 --bench-laps=3
log "batch done"
