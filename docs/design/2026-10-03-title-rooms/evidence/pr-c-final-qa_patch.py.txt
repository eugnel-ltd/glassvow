"""QA-only patch for the #655 PR C final batch (never committed, never in the game).

Applied to a worktree for its QA build and reverted at once after the export.
`--rooms-bench` seeds a saved run (Act II, or Act I with `--bench-act=0`) and a
Vigil part-way in the Development profile, shows the title and attaches
tools/bench_rooms.gd's Bench (copied to tools/qa_rooms_bench.gd, a name the
export keeps), which taps each room's word and its Return through the engine's
own input path and writes one row per drawn frame to user://bench_rooms.jsonl.
The control build (main) has no hearth hall: its copy tours Settings, How to
Play and Credits only.

Arguments: --qa-nonce= (echoed in every row and the report), --bench-laps=,
--bench-title-rest=, --warm-arm=full|build|keep|none:
  full   RoomWarm as built;
  build  RoomWarm builds and frees each room but neither shapes, draws nor
         samples pipelines (#675's next measure for the 0.92 s frame);
  keep   as build, but the rooms built are never freed (is it the free?);
  none   no RoomWarm at all.
Every warm-phase row also carries RoomWarm's stage, the frame's process and
draw segments, pipeline compilations and video memory, and the report drawn
over the screen once the bench is done lists every warm frame over 33 ms with
them, so a long frame says what it was made of. Video memory is reported at the
landing, after the warms, after the title's rest and at the end.

usage: qa_patch.py <worktree> <build-label> [--control] | <worktree> --revert"""
import os
import shutil
import sys

HERE = os.path.dirname(os.path.abspath(__file__))
W = sys.argv[1]
FILES = ["application/main.gd", "presentation/ui/room_warm.gd"]
BENCH = os.path.join(W, "tools/qa_rooms_bench.gd")


def backup(rel):
    return os.path.join(W, rel) + ".qa-backup"


if sys.argv[2] == "--revert":
    for rel in FILES:
        if os.path.exists(backup(rel)):
            shutil.copyfile(backup(rel), os.path.join(W, rel))
            os.remove(backup(rel))
    for path in (BENCH, BENCH + ".uid"):
        if os.path.exists(path):
            os.remove(path)
    sys.exit(0)

LABEL = sys.argv[2]
CONTROL = "--control" in sys.argv[3:]
for rel in FILES:
    assert not os.path.exists(backup(rel)), "a QA patch is already applied"
for rel in FILES:
    shutil.copyfile(os.path.join(W, rel), backup(rel))


def edit(rel, pairs, text=None):
    path = os.path.join(W, rel) if text is None else None
    s = open(path).read() if text is None else text
    for a, b in pairs:
        assert s.count(a) == 1, (rel, a[:90], s.count(a))
        s = s.replace(a, b)
    if text is None:
        open(path, "w").write(s)
    return s

func _qa_rooms_bench(nonce: String, act: int, laps: int, title_rest: float, now: bool) -> void:
	var script: GDScript = load("res://tools/qa_rooms_bench.gd") as GDScript
	script.get("Bench").call("seed_saved_run", act)

func _qa_warm_stage() -> String:

## QA only: Back to the Road timed as R1.1 times it (#671): from the tap to a
## drawn, interactive map. After the rooms' tour (the title rested, the hall
## built ahead), or with `--continue-now` the moment the launch rite lands.
var _qa_continue: Array[String] = []


func _qa_continue_timed(when: String) -> void:
	var title: TitleScreen = _choice_screen as TitleScreen
	if title == null:
		_qa_continue.append("continue %s: no title" % when)
		return
	var ahead: bool = get("_vigil_ahead") != null
	var tap: int = Time.get_ticks_usec()
	title._on_lantern()
	var worst: float = 0.0
	var last: int = tap
	for i: int in range(1200):
		await get_tree().process_frame
		var now: int = Time.get_ticks_usec()
		worst = maxf(worst, float(now - last) / 1000.0)
		last = now
		var screen: WorldMapScreen = _map_screen
		if screen != null and is_instance_valid(screen) and screen.is_inside_tree() \\
				and not screen.landscape_pending() and not screen.layout_pending():
			break
	await RenderingServer.frame_post_draw
	_qa_continue.append("continue %s: ready %.1f ms, worst frame %.1f ms, hall built ahead %s" % [
		when, float(Time.get_ticks_usec() - tap) / 1000.0, worst, str(ahead)])


func _qa_after_tour(bench: Node) -> void:
	await get_tree().create_timer(2.0).timeout
	await _qa_continue_timed("after the tour")
	await get_tree().create_timer(1.0).timeout
	_qa_bench_report(bench)


func _qa_continue_at_landing(bench: Node) -> void:
	while true:
		await get_tree().process_frame
		var title: TitleScreen = _choice_screen as TitleScreen
		if title != null and (title.rite == null or title.rite.is_done()):
			break
	await _qa_continue_timed("as the rite lands")
	await get_tree().create_timer(1.0).timeout
	_qa_bench_report(bench)
'''
