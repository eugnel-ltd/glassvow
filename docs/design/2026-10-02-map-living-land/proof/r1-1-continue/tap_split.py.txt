"""Splits the Continue tap's time to a drawn interactive map by QA marks:
tap -> continue_run (flare and flood until covered), -> bind_journey (restore
and compile), -> built (the land's build ends), -> ready (drawn and
interactive); per launch label prefix. usage: tap_split.py <rows.jsonl> <prefix>..."""
import json, sys, statistics as st
launches = {}
for l in open(sys.argv[1]):
    if l.strip():
        r = json.loads(l); launches.setdefault(r["launch"], {})[r["probe"]] = r
for prefix in sys.argv[2:]:
    rows = []
    for name, by in sorted(launches.items()):
        if not (name.startswith(prefix) and name[len(prefix):].isdigit()) or "continue" not in by:
            continue
        c = by["continue"]; m = c["marks"]; tap = c["tap_us"]
        def t(label, last=False):
            h = [x[0] for x in m if x[1] == label or x[1].startswith(label + " ")]
            return (h[-1] if last else h[0]) if h else None
        cr, bj, built = t("continue_run"), t("bind_journey"), t("built", last=True)
        f = lambda a, b: round((a - b) / 1000, 1) if a and b else None
        rows.append((name, f(cr, tap), f(bj, cr), f(built, tap), c["built_ms"], c["ready_ms"]))
    print("== %s  (tap->continue_run, continue_run->bind, tap->land built, tap->map built, tap->ready)" % prefix)
    for r in rows:
        print("  %-16s %s" % (r[0], r[1:]))
    cols = list(zip(*[r[1:] for r in rows]))
    print("  medians %s" % [round(st.median([v for v in c if v is not None]), 1) if any(v is not None for v in c) else None for c in cols])
