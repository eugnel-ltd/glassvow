shader_type canvas_item;
// Scratch concept C: the painted sky band under the HUD. A night gradient, a
// moon, the act's backdrop ridge sliding at a parallax fraction of the pan,
// fading into the land at its foot. 1 sampler.
uniform sampler2D ridge : source_color, filter_linear;
uniform float pan = 0.0;
uniform vec3 top : source_color = vec3(0.03, 0.03, 0.09);
uniform vec3 low : source_color = vec3(0.20, 0.10, 0.18);
void fragment() {
	vec2 uv = UV;
	vec3 sky = mix(top, low, smoothstep(0.0, 0.85, uv.y));
	vec2 moon = vec2(0.18 - pan * 0.02, 0.32);
	float m = length((uv - moon) * vec2(3.2, 1.0));
	sky += vec3(0.9, 0.85, 0.75) * (1.0 - smoothstep(0.035, 0.04, m)) + vec3(0.25, 0.2, 0.3) * (1.0 - smoothstep(0.04, 0.22, m)) * 0.5;
	vec2 ruv = vec2(fract(uv.x * 0.8 + pan * 0.15), 0.18 + uv.y * 0.82);
	vec4 r = texture(ridge, ruv);
	vec3 c = mix(sky, r.rgb * 0.7, r.a);
	float fade = 1.0 - smoothstep(0.7, 1.0, uv.y);
	COLOR = vec4(c, fade);
}
