extends SceneTree
## Lane R3.2: the floor's pipeline warm-up drawn first (as the title's prime()
## does), then the game booted onto the map; prints the floor bake's frames.
## Without --warm, the map boots at once (the cold path).
func _initialize() -> void:
	MapScene.lean_override = 1
	_run.call_deferred()

func _run() -> void:
	if "--warm" in OS.get_cmdline_user_args():
		var warm: GDScript = load("res://presentation/map/landscape/floor_warm.gd")
		warm.call("warm")
		for i: int in range(8):
			await process_frame
	change_scene_to_file("res://application/main.tscn")
	var screen: WorldMapScreen = null
	for i: int in range(6000):
		await process_frame
		var main: Node = current_scene
		if main == null:
			continue
		var found: Variant = main.get("_map_screen")
		if found is WorldMapScreen and is_instance_valid(found) and not found.landscape_pending():
			screen = found
			break
	var land: MapJourneyLandscape = screen._map_scene.journey_landscape()
	var bake: Dictionary = land.forest_floor.timings_ms.get("bake", {})
	print("WARMCHECK ", JSON.stringify({"warm": "--warm" in OS.get_cmdline_user_args(), "frames_ms": bake.get("frames_ms"), "steps_ms": bake.get("steps_ms"), "wall_ms": bake.get("wall_ms")}))
	quit(0)
