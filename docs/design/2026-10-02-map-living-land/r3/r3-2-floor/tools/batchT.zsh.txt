#!/bin/zsh
# Lane R3.2 revision, batch T: the title path on the iPad 8 (QA app only,
# under the device lock). A seed launch stores an Act I run; then each title
# launch boots to the title (whose prime() draws the floor's warm-up on frame
# 0), waits for the land's prefetch and paced bake, presses Back to the Road
# and times the open. Order: main, the branch's first launch after its
# install (its changed floor shaders not yet compiled), the branch again
# (compiled), main again.
set -u
source ${0:A:h}/common.zsh
OUT=$R/device/batchT
$FLOW/ipad_lock.sh acquire map-r32
trap "$FLOW/ipad_lock.sh release map-r32; echo \"== lock released \$(date +%T)\"" EXIT
UDID=$(mt_udid); [[ -n "$UDID" ]] || { echo "no iPad"; exit 1; }
tt_launch() {  # label wait args...
  local label=$1 wait=$2 i; shift 2
  _mt_start $label "$@"
  [[ -n "$MT_PID" ]] || { echo "== $label: launch failed"; return 1; }
  sleep $wait
  # (As run, the line here removed this launch's own stale row file from the batch's
  # scratch output folder. This copy leaves it out: no committed tool deletes anything.)
  for i in $(seq 1 20); do
    timeout 300 xcrun devicectl device copy from --device $UDID --domain-type appDataContainer \
      --domain-identifier $MT_QA --source Documents/title_probe.jsonl --destination $OUT/$label.jsonl >/dev/null 2>&1
    if grep -q "$MT_NONCE" $OUT/$label.jsonl 2>/dev/null && grep -q '"probe":"\(open\|seed\)"' $OUT/$label.jsonl; then
      echo "== $label nonce $MT_NONCE: rows fresh ($(grep -o '"build":"[^"]*"' $OUT/$label.jsonl | head -1))"; return 0
    fi
    sleep 5
  done
  echo "== $label nonce $MT_NONCE: rows MISSING"; return 1
}
echo "== batch T start $(date +%T)"
install_build ctl $CTL_IPA || exit 1
tt_launch ctl-seed 40 --map --seed=1 --map-steps=2 --title-trace-probe=seed
tt_launch ctl-title1 50 --title-trace-probe=title
install_build dev $DEV_IPA || exit 1
tt_launch dev-title1 60 --title-trace-probe=title
tt_launch dev-title2 50 --title-trace-probe=title
tt_launch dev-title3 50 --title-trace-probe=title
install_build ctl2 $CTL_IPA || exit 1
tt_launch ctl-title2 50 --title-trace-probe=title
echo "== batch T done $(date +%T)"
