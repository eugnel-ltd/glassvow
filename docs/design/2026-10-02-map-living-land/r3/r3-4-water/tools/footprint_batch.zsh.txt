#!/bin/zsh
# Lane R3.4 batch W3 (scratch; QA app only, under the shared lock, taken only when free):
# the peak physical footprint, main then the branch, one install each. Each attempt
# launches the boot-and-rests launch, then records Activity Monitor; a recording that
# the device cuts short ("Device disconnected") is retried with a fresh launch, up to 4.
set -u
L=/Users/jamesto/Coding/glassvow/.claude/worktrees/map-r34-lane
FLOW=/Users/jamesto/Coding/glassvow/.claude/worktrees/device13-flow
LOCK=$FLOW/ipad.lock; OWNER=map-r34
OUT=$L/device/${BATCH:-batchW3}; mkdir -p $OUT
source /Users/jamesto/Coding/glassvow/.claude/worktrees/map-r34/tools/map_trace/device.zsh
while ! mkdir $LOCK 2>/dev/null; do echo "== lock held by '$(cat $LOCK/owner 2>/dev/null)'; waiting $(date +%T)"; sleep 60; done
echo $OWNER > $LOCK/owner; echo "== lock taken $(date +%T)"
trap '[[ "$(cat $LOCK/owner 2>/dev/null)" == $OWNER ]] && rm -rf $LOCK && echo "== lock released $(date +%T)"' EXIT
UDID=$(mt_udid); [[ -n "$UDID" ]] || { echo "no iPad"; exit 1; }
for b in ${BUILDS:-ctl dev}; do
  touch $LOCK/owner
  build=$(cat $L/ipa/$b.build)
  echo "== install $b ($build) $(date +%T)"
  timeout 300 zsh -c "source /Users/jamesto/Coding/glassvow/.claude/worktrees/map-r34/tools/map_trace/device.zsh; UDID=$UDID; OUT=$OUT; mt_install $L/ipa/$b.ipa" || { echo "install failed"; exit 1; }
  for a in 1 2 3 4; do
    lab=$b-a$a
    _mt_start $lab --map --map-trace-probe --map-lean --seed=1 --shape=pad-landscape --map-steps=2 --holds=cadence,live --probe-view=journey,river,close,whole
    [[ -n "$MT_PID" ]] || { echo "== $lab: launch failed"; continue; }
    echo $MT_PID > $OUT/$lab.pid
    sleep 3
    xcrun devicectl device info ddiServices --device $UDID >/dev/null 2>&1
    timeout 300 xcrun xctrace record --device $UDID --template 'Activity Monitor' --all-processes \
      --time-limit 140s --output $OUT/$lab-am.trace --no-prompt 2>&1 | mt_redact > $OUT/$lab-am.log
    _mt_rows $lab
    if grep -q 'time limit' $OUT/$lab-am.log && xcrun xctrace export --input $OUT/$lab-am.trace --toc >/dev/null 2>&1; then echo "== $lab: full recording"; break; fi
    echo "== $lab: recording cut short: $(grep -i disconnect $OUT/$lab-am.log | head -1)"
  done
done
echo "== done $(date +%T)"
