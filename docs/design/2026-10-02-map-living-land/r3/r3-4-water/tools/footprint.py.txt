#!/usr/bin/env python3
"""Lane R3.4 (scratch): the peak physical footprint of one process from an
Activity Monitor trace's live table. Usage: footprint.py <live.xml> <process-substring> [pid]"""
import sys, xml.etree.ElementTree as ET
path, name = sys.argv[1], sys.argv[2]
pid = sys.argv[3] if len(sys.argv) > 3 else None
root = ET.parse(path).getroot()
schema = root.find('.//schema')
cols = [c.find('mnemonic').text for c in schema.findall('col')]
ids = {}
def value(el):
    if 'ref' in el.attrib:
        el = ids[el.attrib['ref']]
    return el
series = []
for row in root.iter('row'):
    cells = list(row)
    for c in row.iter():
        if 'id' in c.attrib:
            ids[c.attrib['id']] = c
    rec = {}
    for col, cell in zip(cols, cells):
        rec[col] = value(cell)
    proc = rec.get('process')
    if proc is None:
        continue
    fmt = proc.attrib.get('fmt', '')
    if name not in fmt or (pid and f'({pid})' not in fmt):
        continue
    fp = rec.get('memory-physical-footprint')
    if fp is None or fp.text is None:
        continue
    series.append(int(fp.text))
if not series:
    print("no rows"); sys.exit(1)
mib = [s / 1048576 for s in series]
print(f"samples {len(mib)} peak {max(mib):.1f} MiB last {mib[-1]:.1f} MiB")
