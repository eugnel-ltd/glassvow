#!/usr/bin/env python3
"""Red share and red hue bands on a capture against the owner's target
(lane R3.1b tint pass, scratch).

red:        hue 340-14, s >= 0.72, v >= 0.12, share of the safe frame (the
            README's red-share mask; the target measures 11.3%).
bands:      of the saturated reds (s >= 0.65, v >= 0.25, hue 345-15), the
            share in 345-355 (crimson), 355-5 (red) and 5-15 (scarlet to
            vermilion): the designer's hue bands.
sat_red:    those saturated reds as a share of the safe frame.
orange:     hue 14-45, s >= 0.72, v >= 0.12 (saturated orange and amber).
crimson:    hue 340-360, s >= 0.5, v >= 0.12.
wood_*:     with -magenta.png beside: of the woodland's own pixels, red
            (340-14, s >= .5, v >= .08), orange, other; and the woodland's
            median value (v) and its share brighter than v 0.35.
Usage: bands.py <capture.png> | --target <png> | --jpg <device.jpg>
"""
import json, sys
import numpy as np
from PIL import Image


def hsv(rgb):
    r, g, b = rgb[..., 0], rgb[..., 1], rgb[..., 2]
    mx = rgb.max(-1); mn = rgb.min(-1); d = mx - mn
    h = np.zeros_like(mx); m = d > 1e-6
    rr = m & (mx == r); gg = m & (mx == g) & ~rr; bb = m & ~rr & ~gg
    h[rr] = ((g - b)[rr] / d[rr]) % 6
    h[gg] = (b - r)[gg] / d[gg] + 2
    h[bb] = (r - g)[bb] / d[bb] + 4
    return h * 60.0, np.where(mx > 1e-6, d / np.maximum(mx, 1e-6), 0), mx


def load(p, box=None):
    im = Image.open(p).convert('RGB')
    if box:
        im = im.crop(box)
    return np.asarray(im).astype(float) / 255.0


def measure(img, wood=None):
    h, s, v = hsv(img)
    n = h.size
    red = ((h <= 14) | (h >= 340)) & (s >= 0.72) & (v >= 0.12)
    orange = (h > 14) & (h <= 45) & (s >= 0.72) & (v >= 0.12)
    crimson = (h >= 340) & (s >= 0.5) & (v >= 0.12)
    sat = (s >= 0.65) & (v >= 0.25)
    b1 = sat & (h >= 345) & (h < 355)
    b2 = sat & ((h >= 355) | (h < 5))
    b3 = sat & (h >= 5) & (h < 15)
    tot = max(int(b1.sum() + b2.sum() + b3.sum()), 1)
    out = {
        'red': round(float(red.sum()) / n, 4),
        'sat_red': round(float(tot) / n, 4),
        'b345_355': round(float(b1.sum()) / tot, 3),
        'b355_5': round(float(b2.sum()) / tot, 3),
        'b5_15': round(float(b3.sum()) / tot, 3),
        'orange': round(float(orange.sum()) / n, 4),
        'crimson': round(float(crimson.sum()) / n, 4),
        'v_median': round(float(np.median(v)), 3),
    }
    if wood is not None:
        m = wood
        k = max(int(m.sum()), 1)
        wr = ((h <= 14) | (h >= 340)) & (s >= 0.5) & (v >= 0.08)
        wo = (h > 14) & (h <= 45) & (s >= 0.5) & (v >= 0.08)
        out['wood_red'] = round(float((wr & m).sum()) / k, 3)
        out['wood_orange'] = round(float((wo & m).sum()) / k, 3)
        out['wood_v_median'] = round(float(np.median(v[m])), 3)
        out['wood_bright'] = round(float(((v >= 0.35) & m).sum()) / k, 3)
        out['cover_magenta'] = round(float(m.mean()), 4)
    return out


def main():
    if sys.argv[1] == '--target':
        print(json.dumps(measure(load(sys.argv[2], (0, 70, 1672, 830)))))
        return
    if sys.argv[1] == '--jpg':
        img = load(sys.argv[2])
        H, W, _ = img.shape
        s = H / 820.0
        print(json.dumps(measure(img[int(76 * s):H - int(46 * s), int(42 * s):W - int(42 * s)])))
        return
    path = sys.argv[1]
    img = load(path)
    H, W, _ = img.shape
    l, t, r, b = 42, 76, 42, 46
    ref_h = 390.0 if W / H > 2.0 else 820.0
    sx = sy = H / ref_h
    crop = lambda a: a[int(t * sy):H - int(b * sy), int(l * sx):W - int(r * sx)]
    c = crop(img)
    wood = None
    try:
        mag = crop(load(path.replace('.png', '-magenta.png')))
        wood = ((mag[..., 0] - mag[..., 1]) > 0.25) & ((mag[..., 2] - mag[..., 1]) > 0.25)
    except FileNotFoundError:
        pass
    out = measure(c, wood)
    try:
        key = crop(load(path.replace('.png', '-key.png')))
        hi = key > 0.5
        fams = {'crimson': (1, 0, 0), 'rust_amber': (1, 1, 0), 'conifer': (0, 0, 1),
                'olive_dark': (0, 1, 0), 'red_under': (1, 0, 1), 'fern': (0, 1, 1)}
        h, s, v = hsv(c)
        red = ((h <= 14) | (h >= 340)) & (s >= 0.72) & (v >= 0.12)
        sat = (s >= 0.65) & (v >= 0.25) & ((h >= 345) | (h < 15))
        n = h.size
        fam_out = {}
        for name, (r, g, b) in fams.items():
            m = (hi[..., 0] == bool(r)) & (hi[..., 1] == bool(g)) & (hi[..., 2] == bool(b))
            if wood is not None:
                m &= wood
            k = max(int(m.sum()), 1)
            hh = h[m & sat]
            fam_out[name] = {'area': round(float(m.sum()) / n, 4), 'red_px': round(float((red & m).sum()) / n, 4),
                             'red_of_fam': round(float((red & m).sum()) / k, 3),
                             'v_med': round(float(np.median(v[m])) if m.any() else 0, 3),
                             'b5_15_of_sat': round(float(((hh >= 5) & (hh < 15)).sum()) / max(hh.size, 1), 3),
                             'b345_of_sat': round(float(((hh >= 345) & (hh < 355)).sum()) / max(hh.size, 1), 3)}
        nonwood = ~wood if wood is not None else np.zeros(h.shape, bool)
        fam_out['not_wood'] = {'red_px': round(float((red & nonwood).sum()) / n, 4)}
        out['families'] = fam_out
    except FileNotFoundError:
        pass
    print(json.dumps(out))


if __name__ == '__main__':
    main()
