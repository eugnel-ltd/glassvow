#!/bin/zsh
# try.zsh <name> <variant args...>: one keyed pad-journey capture of a candidate.
T=/private/tmp/claude-501/-Users-jamesto-Coding-glassvow/b76db86a-b79e-49a2-b973-bc87e903afff/scratchpad/tint

python3 - $T/c/$name/$v.png $name $v <<'PY'
import json, subprocess, sys
d = json.loads(subprocess.check_output(['python3', sys.argv[0].replace('-', '') if False else '/private/tmp/claude-501/-Users-jamesto-Coding-glassvow/b76db86a-b79e-49a2-b973-bc87e903afff/scratchpad/tint/bands.py', sys.argv[1]]))

PY
done
