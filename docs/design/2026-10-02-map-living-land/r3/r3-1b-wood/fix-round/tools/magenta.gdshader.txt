shader_type spatial;
// Scratch (lane R3.1b fix round): the wood's cards as flat magenta, for the
// flat-magenta canopy-cover count; the same cut-out as impostor.gdshader.
render_mode unshaded, cull_back, depth_draw_opaque, shadows_disabled;
uniform sampler2D atlas_albedo : source_color, filter_linear_mipmap, repeat_disable;
uniform float scissor = 0.5;
varying vec4 tile_uv;
void vertex() {
	tile_uv = INSTANCE_CUSTOM;
}
void fragment() {
	vec2 uv = tile_uv.xy + clamp(UV, vec2(0.0), vec2(1.0)) * tile_uv.zw;
	ALBEDO = vec3(1.0, 0.0, 1.0);
	ALPHA = texture(atlas_albedo, uv).a;
	ALPHA_SCISSOR_THRESHOLD = scissor;
}
