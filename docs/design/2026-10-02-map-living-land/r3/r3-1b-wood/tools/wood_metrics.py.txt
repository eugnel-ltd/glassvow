#!/usr/bin/env python3
"""Woodland metrics on a capture (lane R3.1b scratch tool, committed as .txt).

cover:  share of the safe frame's pixels the wood's cards change (the capture
        against the same frame with the cards hidden, `-bare.png`).
red:    share of the safe frame's pixels that read crimson/scarlet by a colour
        mask (hue 340-14 degrees, saturation >= 0.72, value >= 0.12; the
        warm brown ground sits at hue 17-25 and saturation 0.55-0.69).
Usage: wood_metrics.py <capture.png> [--safe=l,t,r,b] [--no-cover]
"""
import sys
import numpy as np
from PIL import Image


def hsv(rgb):
    r, g, b = rgb[..., 0], rgb[..., 1], rgb[..., 2]
    mx = rgb.max(-1); mn = rgb.min(-1); d = mx - mn
    h = np.zeros_like(mx)
    m = d > 1e-6
    rr = m & (mx == r); gg = m & (mx == g) & ~rr; bb = m & ~rr & ~gg
    h[rr] = ((g - b)[rr] / d[rr]) % 6
    h[gg] = (b - r)[gg] / d[gg] + 2
    h[bb] = (r - g)[bb] / d[bb] + 4
    h = h * 60.0
    s = np.where(mx > 1e-6, d / np.maximum(mx, 1e-6), 0)
    return h, s, mx


def red_mask(rgb):
    h, s, v = hsv(rgb)
    return ((h <= 14) | (h >= 340)) & (s >= 0.72) & (v >= 0.12)


def main():
    path = sys.argv[1]
    safe = (42, 76, 42, 46)
    cover = True
    for a in sys.argv[2:]:
        if a.startswith("--safe="):
            safe = tuple(int(v) for v in a[7:].split(","))
        if a == "--no-cover":
            cover = False
    img = np.asarray(Image.open(path).convert("RGB")).astype(float) / 255.0
    h, w, _ = img.shape
    l, t, r, b = safe
    crop = img[t:h - b, l:w - r]
    out = {"red": round(float(red_mask(crop).mean()), 4)}
    if cover:
        bare = np.asarray(Image.open(path.replace(".png", "-bare.png")).convert("RGB")).astype(float) / 255.0
        bcrop = bare[t:h - b, l:w - r]
        diff = np.abs(crop - bcrop).max(-1)
        out["cover"] = round(float((diff > 0.06).mean()), 4)
    print(out)


if __name__ == "__main__":
    main()
