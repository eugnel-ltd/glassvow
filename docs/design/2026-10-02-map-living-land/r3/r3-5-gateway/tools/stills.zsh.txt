#!/bin/zsh
# stills.zsh <tag>: the lane's stills in this worktree's current tree (b = branch, m = main).
T=$1; D=/Users/jamesto/Coding/glassvow/.claude/worktrees/map-arch/artifacts/map-arch/stills; C=/Users/jamesto/Coding/glassvow/.claude/worktrees/map-arch/artifacts/map-arch/cap.sh

typeset -A LEAN; LEAN=(pad-landscape --look-lean phone-landscape --look-lean desktop-landscape --look-settle=60)

  done
done
