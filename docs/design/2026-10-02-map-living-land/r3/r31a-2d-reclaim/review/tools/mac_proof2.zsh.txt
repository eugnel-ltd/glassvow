#!/bin/zsh
# Mac proof for the R3.1a review fixes under the A12 Metal condition: control
# (R2, 8815da34) against the head (code 59abff88, at d3f3a22a).
R=<scratch>/map-lane/r31a
M=$R/fix/mac/proof; mkdir -p $M
shot() { $R/bin/mshot.zsh "$@"; }
J=(--map-steps=2)
# Head: three shapes x lean/full x en/zh-Hant.
for shape in phone-landscape pad-landscape desktop-landscape; do
  for prof in lean full; do
    pa=(); [[ $prof == lean ]] && pa=(--map-lean)
    for loc in en zh-Hant; do
      shot head $M/head-$shape-$prof-$loc.png $pa $J --shape=$shape --locale=$loc
    done
  done
done
# A/B at each shape's production profile: grain on, off and Reduce Motion.
for shape in phone-landscape pad-landscape desktop-landscape; do
  pa=(--map-lean); [[ $shape == desktop-landscape ]] && pa=()
  for b in ctl head; do
    [[ $b == head ]] || shot ctl $M/ctl-$shape-on.png $pa $J --shape=$shape
    shot $b $M/$b-$shape-off.png $pa $J --shape=$shape --probe-grain=off
    shot $b $M/$b-$shape-rm.png $pa $J --shape=$shape --probe-rm
  done
done
# Grain over time: 24 frames in a row, pad lean.
for b in ctl head; do
  shot $b $M/seq-$b.png --map-lean $J --shape=pad-landscape --probe-seq=24
done
# A room over the map (the run's deck), pad lean.
for b in ctl head; do
  shot $b $M/$b-pad-deck.png --map-lean $J --shape=pad-landscape --probe-overlay=deck
done
# The entrance fade on a painted act (Act II) and on Act I: modulate 1 and 0.5.
for b in ctl head; do
  for a in 1 0.5; do
    shot $b $M/$b-pad-act2-fade$a.png --map-lean --act=1 --shape=pad-landscape --probe-fade=$a
    shot $b $M/$b-pad-act1-fade$a.png --map-lean $J --shape=pad-landscape --probe-fade=$a
  done
done
echo "mac proof done"
