#!/bin/zsh
# Lane R3.3 (scratch): the atlas split's lever, proven on the Mac before integrating. The woodland
# albedo atlas as one 2048x1776 page (the texture importer, as shipped) against the same picture as
# a two-layer texture array (VRAM-compressed, mipmapped; the importer makes each 888-row page a
# power of two, 2048x1024, as a VRAM-compressed array's layers must be), each loaded alone in its
# own process on the loader's threads (tex_cost.gd), interleaved, three times each; the normals too.
# Runs each launch only under load 50. Writes only $M/final/split-proof/.
M=/Users/jamesto/Coding/glassvow/.claude/worktrees/map-lane
E=$M/exp-split
F=$M/final/split-proof; mkdir -p $F
source $M/tools/glock.zsh
wait_load() { while (( $(uptime | sed 's/.*averages: //' | cut -d' ' -f1 | cut -d. -f1) >= 50 )); do sleep 30; done; }
n=0
for rep in 1 2 3; do
  for what in single pages single-normal pages-normal; do
    n=$((n+1))
    wait_load
    glock_acquire
    echo "== $n $what $(date +%T) load: $(uptime | sed 's/.*averages: //')"
    GODOT_MTL_DISABLE_ARGUMENT_BUFFERS=1 timeout 300 godot --path $E --rendering-driver metal --rendering-method mobile \
      --position 1400,900 --resolution 320x240 -s $M/tools/tex_cost.gd -- --then=res://$what.png > $F/$n-$what.log 2>&1
    echo "   exit $?"; glock_release; grep TEXCOST $F/$n-$what.log | sed 's/^/   /'
  done
done
echo "== done $(date +%T)"
