extends SceneTree
@warning_ignore_start("unsafe_call_argument", "unsafe_cast", "untyped_declaration", "inferred_declaration", "unsafe_method_access", "unsafe_property_access", "return_value_discarded")
## SCRATCH PROOF (#657): pile states at HUD size (96 x 130 box), 2:1.
##   row 1  the draw pile at 0, 1, 3, 5, 10, 20, 40, 99 cards (thickness law)
##   row 2  discard face-up and loose (top = the last card that landed, a real
##          card face), ashes charred, and the top-menu deck as a mini stack

const Backs: GDScript = preload("res://qa_cards/backs.gd")
const PileStack: GDScript = preload("res://qa_cards/pile_stack.gd")
const BOX: Vector2 = Vector2(96.0, 130.0)


func _initialize() -> void:
	root.content_scale_size = Vector2i(0, 0)
	root.content_scale_factor = 2.0
	CardView.oversample = 4.0
	process_frame.connect(_run, CONNECT_ONE_SHOT)


func _lbl(host: Control, t: String, at: Vector2) -> void:
	var l: Label = Label.new()
	l.text = t
	l.position = at
	l.add_theme_font_size_override("font_size", 10)
	l.add_theme_color_override("font_color", Color(0.85, 0.82, 0.74))
	host.add_child(l)


func _pile(host: Control, tex: Texture2D, at: Vector2, n: int, box: Vector2 = BOX) -> Control:
	var ps: Control = PileStack.new()
	ps.set("tex", tex)
	ps.position = at
	ps.size = box
	var w: float = box.x * 0.86
	ps.set("card_px", Vector2(w, w * CardView.CARD_H / CardView.CARD_W))
	ps.set("count", n)
	host.add_child(ps)
	return ps


func _run() -> void:
	var host: Control = Control.new()
	host.set_anchors_preset(Control.PRESET_FULL_RECT)
	var bg: TextureRect = TextureRect.new()
	bg.texture = load("res://assets/art/stage/act1-ledge.png") as Texture2D
	bg.expand_mode = TextureRect.EXPAND_IGNORE_SIZE
	bg.stretch_mode = TextureRect.STRETCH_KEEP_ASPECT_COVERED
	bg.set_anchors_preset(Control.PRESET_FULL_RECT)
	bg.modulate = Color(0.6, 0.55, 0.55)
	host.add_child(bg)
	root.add_child(host)
	var back: Dictionary = await Backs.bake(host, OS.get_environment("BACK") if OS.get_environment("BACK") != "" else "rose")
	var tex: Texture2D = Backs.card_region(back)
	var counts: Array[int] = [0, 1, 3, 5, 10, 20, 40, 99]
	for i: int in range(counts.size()):
		var at: Vector2 = Vector2(16.0 + float(i) * 108.0, 22.0)
		_pile(host, tex, at, counts[i])
		_lbl(host, "draw · %d" % counts[i], at + Vector2(20.0, BOX.y + 2.0))
	# A real card face for the discard's top: bake one CardView's stage.
	var content: ContentDB = ContentDB.load_full()
	var face: CardView = CardView.new(CardInst.new(7, &"eclipseSlash"), content.cards.get("eclipseSlash", {}), 2)
	face.position = Vector2(-3000, -3000)
	host.add_child(face)
	for _i: int in range(6):
		await process_frame
	var fimg: Image = face._stage.get_texture().get_image()
	fimg.generate_mipmaps()
	var ftex: AtlasTexture = AtlasTexture.new()
	ftex.atlas = ImageTexture.create_from_image(fimg)
	var s: float = CardView.oversample
	ftex.region = Rect2(Vector2(24.0, 24.0) * s, Vector2(152.0, 216.0) * s)
	var y2: float = 22.0 + BOX.y + 30.0
	var d: Control = _pile(host, tex, Vector2(16.0, y2), 7)
	d.set("face_up_top", ftex)
	d.set("loose", 1.0)
	_lbl(host, "discard · 7 · face-up", Vector2(16.0, y2 + BOX.y + 2.0))
	var a: Control = _pile(host, tex, Vector2(124.0, y2), 4)
	a.set("burnt", 1.0)
	a.set("loose", 0.6)
	_lbl(host, "ashes · 4 · charred", Vector2(124.0, y2 + BOX.y + 2.0))
	# The top-menu deck: a 40x56 stack where the icon is today.
	var deck_counts: Array[int] = [10, 25, 40]
	for i: int in range(deck_counts.size()):
		var at: Vector2 = Vector2(270.0 + float(i) * 130.0, y2 + 40.0)
		var p: Control = _pile(host, tex, at, deck_counts[i], Vector2(44.0, 64.0))
		p.set("card_px", Vector2(36.0, 51.0))
		var n: Label = Label.new()
		n.text = str(deck_counts[i])
		n.position = at + Vector2(6.0, 18.0)
		n.add_theme_font_size_override("font_size", 16)
		n.add_theme_color_override("font_outline_color", Color.BLACK)
		n.add_theme_constant_override("outline_size", 5)
		host.add_child(n)
		_lbl(host, "top-menu deck · %d" % deck_counts[i], at + Vector2(-20.0, 68.0))
	for _i: int in range(4):
		await process_frame
	var out: String = OS.get_environment("OUT")
	if out != "":
		root.get_texture().get_image().save_png(out)
		print("saved ", out)
	quit()
