extends SceneTree
@warning_ignore_start("unsafe_call_argument", "unsafe_cast", "untyped_declaration", "inferred_declaration", "unsafe_method_access", "unsafe_property_access", "return_value_discarded")
## SCRATCH (#657 PR 5): frame bursts of the piles in a live bench fight,
## through the real paths (the drag gestures, end turn, the reshuffle the
## domain makes, a pile tap).
##   godot --path <wt> --position 60,60 -s <this> -- --fight=duskfang --kind=elite \
##     --seed=1 --vp=WxH --out=<dir>/<prefix> [--scenes=play,strike,sweep,reshuffle]
##     [--slow=0.1] [--dt=0.04] [--rm=1] [--skip=1]
## Each frame is the root viewport, kept in memory and written at the end with
## its game time in the name: <prefix>_<scene>_<ms>.png

var cfg: Dictionary = {"out": "/tmp/burst", "scenes": "play,strike,sweep,reshuffle",
	"slow": "0.1", "dt": "0.04", "rm": "0", "settle": "6", "skip": "0"}
var main: Node = null
var _frames: Array = []


func _initialize() -> void:
	for arg: String in OS.get_cmdline_user_args():
		for k: String in cfg.keys():
			if arg.begins_with("--%s=" % k):
				cfg[k] = arg.trim_prefix("--%s=" % k)
	main = (load("res://application/main.tscn") as PackedScene).instantiate()
	root.add_child(main)
	_run.call_deferred()


func _wait(seconds: float) -> void:
	var until: int = Time.get_ticks_msec() + int(seconds * 1000.0)
	while Time.get_ticks_msec() < until:
		await process_frame


func _tap() -> void:
	var ev: InputEventMouseButton = InputEventMouseButton.new()
	ev.button_index = MOUSE_BUTTON_LEFT
	ev.pressed = true
	ev.position = Vector2(root.size) * Vector2(0.5, 0.3)
	Input.parse_input_event(ev)
	var up: InputEventMouseButton = ev.duplicate()
	up.pressed = false
	Input.parse_input_event(up)
	print("TAP")


func _burst(scene: String, game_seconds: float, tap_at: float = -1.0) -> void:
	var slow: float = float(cfg["slow"])
	var dt: float = float(cfg["dt"])
	Engine.time_scale = slow
	var t0: int = Time.get_ticks_usec()
	var game_t: float = 0.0
	var next: float = 0.0
	var tapped: bool = false
	while game_t <= game_seconds + 0.0001:
		await process_frame
		game_t = float(Time.get_ticks_usec() - t0) / 1e6 * slow
		if tap_at >= 0.0 and not tapped and game_t >= tap_at:
			tapped = true
			_tap()
		if game_t >= next:
			_frames.append(["%s_%s_%04d" % [cfg["out"], scene, int(round(game_t * 1000.0))],
				root.get_texture().get_image()])
			next += dt
	Engine.time_scale = 1.0
	await _wait(0.6)


func _idle(screen) -> void:
	var idle: int = Time.get_ticks_msec()
	while screen.seq.is_busy() and Time.get_ticks_msec() - idle < 20000:
		await process_frame
	await _wait(0.8)


func _run() -> void:
	await _wait(float(cfg["settle"]))
	if cfg["rm"] == "1":
		Preferences.active.reduce_motion = true
	var screen = main._screen
	var hand = screen._hand
	var hud = screen._hud
	print("BURST shape=%s window=%s rm=%s back=%s" % [str(screen.shape), str(root.size), cfg["rm"], str(CardTurn.back() != null)])
	for scene: String in str(cfg["scenes"]).split(",", false):
		await _idle(screen)
		var cb = screen.game.cb
		print("SCENE %s hand=%d draw=%d discard=%d ashes=%d energy=%d top=%d" % [scene, cb.hand.size(), cb.draw.size(), cb.discard.size(), cb.exhaust.size(), cb.player.energy, hud.discard_top_uid()])
		match scene:
			"play":
				var uid: int = _find(screen, "skill", false)
				if uid < 0:
					print("no plain skill in hand"); continue
				var v = hand.card_view(uid)
				var p0: Vector2 = v.global_centre()
				hand._on_card_pressed_at(uid, p0)
				hand._on_card_moved_to(uid, p0 + Vector2(40, -260))
				await process_frame
				hand._on_card_released_at(uid, p0 + Vector2(40, -260))
				await _burst(scene, 0.9)
			"strike":
				var uid: int = _find(screen, "attack", false)
				if uid < 0:
					print("no attack in hand"); continue
				var v = hand.card_view(uid)
				var p0: Vector2 = v.global_centre()
				hand._on_card_pressed_at(uid, p0)
				hand._on_card_moved_to(uid, p0 + Vector2(0, -60))
				await process_frame
				hand._on_card_released_at(uid, screen._enemy_centre(0))
				await _burst(scene, 2.6)
				print("STRUCK uid=%d top=%d" % [uid, hud.discard_top_uid()])
			"sweep":
				screen._on_end_turn_pressed()
				await _burst(scene, 1.6)
			"reshuffle":
				# End turns until a draw empties the pile and the domain reshuffles.
				for _i: int in range(4):
					if screen.game.cb.draw.size() < 5:
						break
					screen._on_end_turn_pressed()
					await _idle(screen)
				print("RESHUFFLE draw=%d discard=%d" % [screen.game.cb.draw.size(), screen.game.cb.discard.size()])
				screen._on_end_turn_pressed()
				var waited: int = Time.get_ticks_msec()
				while hud._stream == null and Time.get_ticks_msec() - waited < 20000 \
						and not (cfg["rm"] == "1" and _pulsing(hud)):
					await process_frame
				print("STREAM at draw=%d discard=%d" % [hud.pile_count(&"draw"), hud.pile_count(&"discard")])
				await _burst(scene + ("_skip" if cfg["skip"] == "1" else ""), 1.3,
					0.2 if cfg["skip"] == "1" else -1.0)
				print("AFTER draw=%d/%d discard=%d/%d" % [hud.pile_count(&"draw"), screen.game.cb.draw.size(), hud.pile_count(&"discard"), screen.game.cb.discard.size()])
			"inspector":
				screen._show_pile(&"draw")
				await _wait(1.5)
				_frames.append(["%s_%s_%04d" % [cfg["out"], scene, 0], root.get_texture().get_image()])
				# A finger on a card: the one live card under it.
				var grid = screen._inspector_grid
				if grid != null and grid.cards().size() > 0:
					var card = grid.cards()[0]
					card.pointer_changed.emit(card, true)
					await _wait(0.8)
					_frames.append(["%s_%s_%04d" % [cfg["out"], scene, 1], root.get_texture().get_image()])
					print("INSPECTOR cards=%d live=%s" % [grid.cards().size(), str(card.live() != null)])
				screen._close_inspector()
			"idle":
				var draw = hud._draw_pile.stack
				var ash = hud._ashes_pile.stack
				draw._t = 0.0
				ash._t = 0.0
				Engine.time_scale = 1.0
				var t0: int = Time.get_ticks_msec()
				var next: int = 0
				while Time.get_ticks_msec() - t0 < 3300:
					await process_frame
					var el: int = Time.get_ticks_msec() - t0
					if el >= next:
						_frames.append(["%s_%s_%04d" % [cfg["out"], scene, el], root.get_texture().get_image()])
						next += 100 if el < 1500 else 400
	print("BURST writing %d frames" % _frames.size())
	for f: Array in _frames:
		(f[1] as Image).save_png(str(f[0]) + ".png")
	quit()


func _pulsing(hud) -> bool:
	return not hud._draw_pile.stack.modulate.is_equal_approx(Color.WHITE)


func _find(screen, type: String, exhaust: bool) -> int:
	for c in screen.game.cb.hand:
		var d: Dictionary = screen.game.rules.card_data(c)
		if str(d.get("type", "")) == type and bool(d.get("exhaust", false)) == exhaust \
				and screen.game.rules.can_play(screen.game.run, screen.game.cb, c,
					0 if str(d.get("target", "")) == "enemy" else null):
			return c.uid
	return -1
