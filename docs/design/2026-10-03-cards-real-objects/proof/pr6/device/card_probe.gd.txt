## SCRATCH QA PROBE (#657 PR 6, never committed). Attached by the QA patch in
## Main._ready when a `--card-probe=` user argument is given:
##   -- --map --seed=1 --card-probe=nonce=N,mode=map|fight[,warm=1]
## The same probe runs on a build of main (A) and of the branch (B).
##   mode=map    the install's warm-up: the map settles, 1 s of rest.
##   mode=fight  the bench fight (duskfang, elite) from the settled map, as
##               PR 5b's probe ran it: a hand card kindled to the ash; End Turn
##               (turn 1's end: the sweep, the foe, the next deal); then
##               REST s at rest with five cards in hand (phase rest: the frame
##               times and the video memory). Then THE TOUCH (`_touch`): the
##               hand's front card entered as a finger's emulated mouse enters
##               it, held 1 s, left, settled (phases touch, untouch); a play
##               from rest (a defend to the
##               discard, else an attack at the foe; phase play); then End Turns
##               until the draw pile reshuffles (resh while the stream flies,
##               rdeal from its end until the drain is idle); 1.5 s of rest.
## Each frame row: phase, wall ms, start-to-draw, draw, process, root GPU,
## video memory MiB, pipeline compiles, and how many hand cards hold a stage
## (-1 on a build whose cards always do). Rows are held in memory and written at
## the end: no file I/O inside a measured frame.

const MONITORS: Array[int] = [
	Performance.PIPELINE_COMPILATIONS_CANVAS, Performance.PIPELINE_COMPILATIONS_MESH,
	Performance.PIPELINE_COMPILATIONS_SURFACE, Performance.PIPELINE_COMPILATIONS_DRAW,
	Performance.PIPELINE_COMPILATIONS_SPECIALIZATION]
const REST: float = 10.0

var cfg: Dictionary = {}
var main: Main = null
var _rows: FileAccess = null
var _phase: String = "boot"
var _frame: int = 0
var _t_start: int = 0
var _t_pre: int = 0
var _t_post: int = 0
var _frame_rows: PackedStringArray = PackedStringArray()
var _vp: RID
var _marks: Dictionary = {}


static func config() -> Dictionary:
	var out: Dictionary = {"nonce": "none", "mode": "fight", "warm": "0", "quit": "1", "args": ""}
	for arg: String in OS.get_cmdline_user_args():
		if arg.begins_with("--card-probe="):
			for part: String in arg.trim_prefix("--card-probe=").split(",", false):
				var kv: PackedStringArray = part.split("=", true, 1)
				if kv.size() == 2:
					out[kv[0]] = kv[1]
	out["args"] = " ".join(OS.get_cmdline_user_args())
	out["rows"] = "user://card_probe_%s.txt" % out["nonce"]
	return out


func _ready() -> void:
	name = "CardProbe"
	process_mode = Node.PROCESS_MODE_ALWAYS
	cfg = config()
	main = get_parent() as Main
	_vp = get_viewport().get_viewport_rid()
	RenderingServer.viewport_set_measure_render_time(_vp, true)
	_row("PROBE nonce=%s mode=%s args=[%s] engine=%s adapter=%s window=%s build=%s ready_ms=%d" % [
		cfg["nonce"], cfg["mode"], cfg["args"], Engine.get_version_info()["string"],
		RenderingServer.get_video_adapter_name(), str(DisplayServer.window_get_size()),
		str((load("res://qa_cards/build_tag.gd") as GDScript).get_script_constant_map().get("TAG", "?")),
		Time.get_ticks_msec()])
	get_tree().process_frame.connect(_on_frame_start)
	RenderingServer.frame_pre_draw.connect(func() -> void: _t_pre = Time.get_ticks_usec())
	RenderingServer.frame_post_draw.connect(func() -> void: _t_post = Time.get_ticks_usec())
	_t_start = Time.get_ticks_usec()
	_run()


func _row(line: String) -> void:
	var stamped: String = "%s %d %s" % [cfg.get("nonce", "none"), Time.get_ticks_msec(), line]
	print(stamped)
	if _rows == null:
		_rows = FileAccess.open(str(cfg["rows"]), FileAccess.WRITE)
	if _rows != null:
		_rows.store_line(stamped)
		_rows.flush()


## Marks the frame something happened in (the frame whose row comes next).
func _mark(key: String) -> void:
	_marks[key] = _frame
	_frame_rows.append("%s %d MARK %s frame=%d ms=%d" % [cfg.get("nonce", "none"),
		Time.get_ticks_msec(), key, _frame, Time.get_ticks_msec()])


func _staged() -> int:
	var screen = main._screen
	if screen == null or not is_instance_valid(screen):
		return -1
	var n: int = 0
	for uid: int in screen._hand.uids():
		var card: CardView = screen._hand.card_view(uid)
		if not card.has_method("has_stage"):
			return -1
		n += 1 if card.call("has_stage") else 0
	return n


func _on_frame_start() -> void:
	var now: int = Time.get_ticks_usec()
	var wall: float = float(now - _t_start) / 1000.0
	var to_draw: float = float(_t_pre - _t_start) / 1000.0 if _t_pre >= _t_start else -1.0
	var draw: float = float(_t_post - _t_pre) / 1000.0 if _t_post >= _t_pre and _t_pre >= _t_start else -1.0
	var counts: PackedStringArray = PackedStringArray()
	for m: int in MONITORS:
		counts.append(str(int(Performance.get_monitor(m))))
	_frame_rows.append("%s %d F %d %s %.2f %.2f %.2f %.2f %.2f %.1f %s %d" % [cfg.get("nonce", "none"),
		Time.get_ticks_msec(), _frame, _phase, wall, to_draw, draw,
		Performance.get_monitor(Performance.TIME_PROCESS) * 1000.0,
		RenderingServer.viewport_get_measured_render_time_gpu(_vp),
		Performance.get_monitor(Performance.RENDER_VIDEO_MEM_USED) / 1048576.0, ",".join(counts),
		_staged()])
	_frame += 1
	_t_start = now


func _wait(seconds: float) -> void:
	var until: int = Time.get_ticks_msec() + int(seconds * 1000.0)
	while Time.get_ticks_msec() < until:
		await get_tree().process_frame


func _frames(n: int) -> void:
	for _i: int in range(n):
		await get_tree().process_frame


func _busy(screen: CombatScreen) -> bool:
	var hand: HandView = screen._hand
	var flying: bool = not hand._dealing.is_empty() or not hand._leaving.is_empty()
	return screen.seq.is_busy() or flying or _streaming(screen)


func _streaming(screen: CombatScreen) -> bool:
	var stream = screen._hud._stream
	return stream != null and is_instance_valid(stream)


## Waits until the drain is idle and nothing is in the air, plus 15 frames.
## While `resh` is set, the frames are tagged by the reshuffle's progress.
func _idle(screen: CombatScreen, label: String, resh: bool = false) -> void:
	var t0: int = Time.get_ticks_msec()
	var streamed: bool = false
	await get_tree().process_frame
	while (_busy(screen) or (resh and not streamed and Time.get_ticks_msec() - t0 < 1500)) \
			and Time.get_ticks_msec() - t0 < 12000:
		if resh:
			if _streaming(screen):
				streamed = true
				_phase = "resh"
			elif streamed:
				_phase = "rdeal"
		await get_tree().process_frame
	var landed: int = Time.get_ticks_msec() - t0
	await _frames(15)
	var cb: CombatState = screen.game.cb
	_row("%s landed_ms=%d streamed=%s draw=%d discard=%d hand=%d ash=%d hp=%d staged=%d" % [label,
		landed, str(streamed), cb.draw.size(), cb.discard.size(), cb.hand.size(),
		cb.exhaust.size(), cb.player.hp, _staged()])


func _settle() -> void:
	var calm: int = 0
	var waited: int = Time.get_ticks_msec()
	while calm < 30 and Time.get_ticks_msec() - waited < 30000:
		var before: int = Time.get_ticks_usec()
		await get_tree().process_frame
		var map = main._map_screen
		var land_ok: bool = map != null and is_instance_valid(map) and map.is_inside_tree() \
			and not map.landscape_pending()
		calm = calm + 1 if land_ok and float(Time.get_ticks_usec() - before) / 1000.0 < 40.0 else 0
	_row("SETTLED ms=%d" % Time.get_ticks_msec())


func _run() -> void:
	await _settle()
	if cfg["mode"] == "map":
		_phase = "maprest"
		await _wait(1.0)
	else:
		await _fight()
	_phase = "end"
	if _rows != null:
		for line: String in _frame_rows:
			_rows.store_line(line)
	_row("DONE")
	if _rows != null:
		_rows.close()
	if cfg["quit"] == "1":
		get_tree().quit()


func _fight() -> void:
	_phase = "map"
	await _wait(1.0)
	_row("BEFORE_LOAD baked=%s back=%s" % [str(CardTurn.back() != null), CardTurn.wearing()])
	_phase = "load"
	main._start_fight(PackedStringArray(["duskfang"]), "elite")
	await get_tree().process_frame
	_phase = "fight"
	_row("AFTER_LOAD back=%s baked=%s rm=%s coarse=%s" % [CardTurn.wearing(),
		str(CardTurn.back() != null), str(Preferences.active.reduce_motion), str(PointerDevice.coarse())])
	await _wait(3.5)
	var screen: CombatScreen = main._screen
	var cb: CombatState = screen.game.cb
	await _idle(screen, "OPEN")
	var seen: Dictionary = {}
	for c: CardInst in cb.hand:
		seen[c.uid] = true
	_phase = "kindle"
	var uid: int = cb.hand[0].uid
	_row("KINDLE uid=%d ok=%s" % [uid, str(screen.request_kindle(uid))])
	await _idle(screen, "KINDLED")
	_phase = "endturn1"
	screen._on_end_turn_pressed()
	await _idle(screen, "TURN2")
	for c: CardInst in cb.hand:
		seen[c.uid] = true
	_phase = "rest"
	await _wait(1.0 if cfg["warm"] == "1" else REST)
	await _touch(screen)
	_phase = "play"
	var played: int = -1
	for c: CardInst in cb.hand:
		var view: CardView = screen._hand.card_view(c.uid)
		if view != null and view.playable and view.target_kind != "enemy":
			played = c.uid
			break
	var target: Variant = null
	if played < 0:
		for c: CardInst in cb.hand:
			var view: CardView = screen._hand.card_view(c.uid)
			if view != null and view.playable:
				played = c.uid
				target = 0
				break
	if played >= 0:
		var view: CardView = screen._hand.card_view(played)
		_row("PLAY uid=%d id=%s staged=%s ok=%s" % [played, view.card_id,
			str(view.call("has_stage")) if view.has_method("has_stage") else "-",
			str(screen.request_play(played, target))])
		_mark("play")
	await _idle(screen, "PLAYED")
	for k: int in range(2, 6):
		var discard: Array[int] = []
		for c: CardInst in cb.discard + cb.hand:
			discard.append(c.uid)
		var will: bool = cb.draw.size() < 5
		var before_all: bool = true
		for u: int in discard:
			before_all = before_all and seen.has(u)
		_row("ENDTURN %d draw=%d reshuffle_due=%s cards=%d dealt_before=%s" % [k, cb.draw.size(),
			str(will), discard.size(), str(before_all)])
		_phase = "endturn%d" % k
		screen._on_end_turn_pressed()
		await _idle(screen, "TURN%d" % (k + 1), will)
		for c: CardInst in cb.hand:
			seen[c.uid] = true
		if will or cb.over:
			break
	_phase = "rrest"
	await _wait(1.5)


## The first touch after the hand let its stages go. The finger enters the
## front card as a touch's emulated mouse does (CardView.point_at: the card's
## mouse_entered, then the motion that sets the lamp down), and is held 1 s;
## then it leaves (point_away) and the card settles. No tap: on a touch screen
## a tap also lifts the card, and on a desktop it would arm or play it.
func _touch(screen: CombatScreen) -> void:
	var hand: HandView = screen._hand
	var uids: Array[int] = hand.uids()
	var card: CardView = hand.card_view(uids[-1])
	var local: Vector2 = Vector2(CardView.CARD_W * 0.6, CardView.CARD_H * 0.15)
	_row("TOUCH uid=%d id=%s staged_before=%d hovered=%s" % [card.uid, card.card_id,
		_staged(), str(card._hovered)])
	_phase = "touch"
	_mark("touch")
	card.point_at(local)
	_row("TOUCHED live=%s staged=%d" % [str(card._hovered), _staged()])
	await _wait(1.0)
	_phase = "untouch"
	_mark("untouch")
	card.point_away()
	await _wait(2.0)
	_row("UNTOUCHED hovered=%s processing=%s staged=%d" % [str(card._hovered),
		str(card.is_processing()), _staged()])
