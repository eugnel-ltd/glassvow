extends SceneTree
@warning_ignore_start("unsafe_call_argument", "unsafe_cast", "untyped_declaration", "inferred_declaration", "unsafe_method_access", "unsafe_property_access", "return_value_discarded")
## SCRATCH PROOF (#657): three backs beside a real front, live CardViews at
## 2:1, plus the baked pile textures. Env: OUT=<png>, LAMP="x,y,gain".

const Backs: GDScript = preload("res://qa_cards/backs.gd")


func _initialize() -> void:
	root.content_scale_size = Vector2i(0, 0)
	var z: float = float(OS.get_environment("ZOOM")) if OS.get_environment("ZOOM") != "" else 1.0
	root.content_scale_factor = z
	CardView.oversample = 2.0 * z
	var host: Control = Control.new()
	host.set_anchors_preset(Control.PRESET_FULL_RECT)
	var bg: ColorRect = ColorRect.new()
	bg.color = Color(0.06, 0.05, 0.07)
	bg.set_anchors_preset(Control.PRESET_FULL_RECT)
	host.add_child(bg)
	root.add_child(host)
	_run(host)


func _run(host: Control) -> void:
	var content: ContentDB = ContentDB.load_full()
	var ids: Array[String] = ["vault", "rose", "eclipse"]
	var x: float = 40.0
	var front: CardInst = CardInst.new(1, &"eclipseSlash")
	var fdata: Dictionary = content.cards.get("eclipseSlash", {})
	var fv: CardView = CardView.new(front, fdata, 2)
	fv.position = Vector2(x, 60.0)
	host.add_child(fv)
	x += 190.0
	var uid: int = 2
	for id: String in ids:
		var v: CardView = Backs.build(id, uid)
		v.position = Vector2(x, 60.0)
		host.add_child(v)
		x += 190.0
		uid += 1
	var label: Label = Label.new()
	label.text = "front (Eclipse Slash) · vault (aurora) · rose (prism) · eclipse (gilt) — live slabs, room lamp"
	label.position = Vector2(40.0, 20.0)
	host.add_child(label)
	for _i: int in range(12):
		await process_frame
	var out: String = OS.get_environment("OUT")
	if out != "":
		root.get_texture().get_image().save_png(out)
		print("saved ", out)
	quit()
