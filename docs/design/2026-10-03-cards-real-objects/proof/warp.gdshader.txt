shader_type canvas_item;
// SCRATCH PROOF (#657) — option B: a 2.5D card turn on the canvas.
//
// Drawn on CardView's own `display` rect (the stage texture, 200 x 264 card
// px at rest). Each fragment casts the ray the card's own stage camera would,
// into a slab rotated by (pitch, yaw): the front face samples the frozen stage
// texture, the back face the baked back, and a ray that enters through the
// side band paints the stock's cross-section. Exact perspective; one texture
// read per face; no viewport renders while the card turns.

uniform sampler2D back_tex : filter_linear_mipmap, source_color;
uniform float yaw = 0.0;      // degrees about the card's vertical axis
uniform float pitch = 0.0;    // degrees about its horizontal axis
uniform float cam = 748.6;    // the card stage's camera distance, card px
uniform vec2 rect = vec2(200.0, 264.0);
uniform vec2 half_card = vec2(76.0, 108.0);
uniform float thick = 2.5;    // slab thickness, card px (exaggerated in flight)
uniform vec4 edge_col : source_color = vec4(0.62, 0.58, 0.48, 1.0);
uniform float radius = 11.0;

mat3 rot_y(float a) {
	float c = cos(a); float s = sin(a);
	return mat3(vec3(c, 0.0, -s), vec3(0.0, 1.0, 0.0), vec3(s, 0.0, c));
}
mat3 rot_x(float a) {
	float c = cos(a); float s = sin(a);
	return mat3(vec3(1.0, 0.0, 0.0), vec3(0.0, c, s), vec3(0.0, -s, c));
}
float rr(vec2 p, vec2 h, float r) {
	vec2 q = abs(p) - (h - vec2(r));
	return length(max(q, vec2(0.0))) + min(max(q.x, q.y), 0.0) - r;
}

void fragment() {
	vec2 s = (UV - 0.5) * rect;
	s.y = -s.y;
	mat3 R = rot_y(radians(yaw)) * rot_x(radians(pitch));
	mat3 Ri = transpose(R);
	vec3 o = Ri * vec3(0.0, 0.0, cam);
	vec3 d = Ri * normalize(vec3(s, -cam));
	vec4 col = vec4(0.0);
	// The face that looks at the camera.
	bool front = o.z > 0.0;
	float zf = front ? thick * 0.5 : -thick * 0.5;
	float tf = (zf - o.z) / d.z;
	vec3 pf = o + d * tf;
	vec2 uv = vec2((front ? pf.x : -pf.x) / rect.x + 0.5, 0.5 - pf.y / rect.y);
	vec4 face = front ? texture(TEXTURE, uv) : texture(back_tex, uv);
	if (uv.x < 0.0 || uv.x > 1.0 || uv.y < 0.0 || uv.y > 1.0) { face = vec4(0.0); }
	// Light falls off as the card turns away from the room lamp — the one cue
	// a frozen texture cannot give itself. Cheap, and only while it turns.
	vec3 n = R * vec3(0.0, 0.0, front ? 1.0 : -1.0);
	float lit = mix(0.62, 1.0, clamp(n.z, 0.0, 1.0));
	face.rgb *= lit;
	// The side band: a ray that enters the slab through its rim.
	vec3 inv = 1.0 / d;
	vec3 hb = vec3(half_card, thick * 0.5);
	vec3 t1 = (-hb - o) * inv;
	vec3 t2 = (hb - o) * inv;
	vec3 tmn = min(t1, t2);
	vec3 tmx = max(t1, t2);
	float tin = max(max(tmn.x, tmn.y), tmn.z);
	float tout = min(min(tmx.x, tmx.y), tmx.z);
	vec4 side = vec4(0.0);
	if (tin < tout && tin > 0.0 && tin == max(tmn.x, tmn.y)) {
		vec3 ps = o + d * tin;
		if (rr(ps.xy, half_card, radius) < 0.6) {
			float shade = tmn.x > tmn.y ? 0.85 : 0.65;
			side = vec4(edge_col.rgb * shade, 1.0);
		}
	}
	col = mix(side, vec4(face.rgb, 1.0), face.a);
	col.a = max(face.a, side.a);
	COLOR = col;
}
