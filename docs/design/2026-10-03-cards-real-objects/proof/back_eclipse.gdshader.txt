shader_type canvas_item;
// SCRATCH PROOF (issue #657) — the "Eclipse" card back: an engine-turned gold
// field round a black sun and its corona. Card px (152 x 216); the surface
// recipe (gilt) makes the gold behave as leaf under the lamp.

uniform vec2 card = vec2(152.0, 216.0);
uniform vec4 ink : source_color = vec4(0.028, 0.026, 0.04, 1.0);
uniform vec4 gold : source_color = vec4(0.95, 0.76, 0.31, 1.0);
uniform vec4 gold_dim : source_color = vec4(0.55, 0.40, 0.13, 1.0);
uniform vec4 ember : source_color = vec4(1.0, 0.55, 0.18, 1.0);

float rr_sdf(vec2 p, vec2 half_s, float r) {
	vec2 q = abs(p) - (half_s - vec2(r));
	return length(max(q, vec2(0.0))) + min(max(q.x, q.y), 0.0) - r;
}

float line(float d, float w) {
	return 1.0 - smoothstep(w * 0.5, w * 0.5 + 0.8, abs(d));
}

void fragment() {
	vec2 cp = UV * card;
	vec2 p = cp - card * 0.5;
	float r = length(p);
	float a = atan(p.y, p.x);
	vec3 col = ink.rgb;

	// GUILLOCHE — a rosette of displaced rings, the engine-turned field of a
	// banknote. Two families crossing make the moire that reads as metal.
	float g1 = sin(r * 0.95 + 3.2 * sin(a * 9.0));
	float g2 = sin(r * 0.95 - 3.2 * sin(a * 9.0 + 0.35));
	float w = fwidth(g1) * 1.3 + 0.02;
	float field = max(1.0 - smoothstep(0.0, w, abs(g1)), 1.0 - smoothstep(0.0, w, abs(g2)));
	float fade = smoothstep(30.0, 44.0, r) * (1.0 - smoothstep(-14.0, -9.0, rr_sdf(p, card * 0.5, 11.0)));
	col = mix(col, gold_dim.rgb, field * 0.55 * fade);

	// THE ECLIPSE — a black disc, the corona round it, the diamond-ring bead.
	float disc = 22.0;
	float rays = 0.55 + 0.45 * sin(a * 23.0) * sin(a * 7.0 + 1.3);
	float corona = exp(-max(r - disc, 0.0) / (5.0 + 7.0 * rays)) * step(disc, r);
	col += ember.rgb * corona * 0.55 + gold.rgb * corona * 0.6;
	col = mix(col, gold.rgb * 1.15, line(r - disc - 0.8, 1.8));
	col = mix(col, ink.rgb * 0.4, 1.0 - smoothstep(disc - 1.0, disc, r));
	vec2 bead = p - vec2(cos(-0.8), sin(-0.8)) * (disc + 0.5);
	col += vec3(1.0, 0.93, 0.75) * exp(-dot(bead, bead) / 9.0) * 1.4;
	// a fine engraved ring of 60 ticks, like a dial
	float tick = line(fract(a / TAU * 60.0) - 0.5, 0.18) * line(r - 36.0, 4.0);
	col = mix(col, gold.rgb, tick * 0.9);
	col = mix(col, gold.rgb, line(r - 33.5, 0.9));
	col = mix(col, gold.rgb, line(r - 38.5, 0.9));

	// the frame: double gold rule, inner bead line, corner rosettes
	float d = rr_sdf(p, card * 0.5, 11.0);
	col = mix(col, gold.rgb, line(d + 6.5, 1.8));
	col = mix(col, gold_dim.rgb, line(d + 10.0, 0.8));
	vec2 corner = abs(p) - (card * 0.5 - vec2(17.0, 17.0));
	float cr = length(corner);
	float petals = 4.2 + 1.6 * cos(atan(corner.y, corner.x) * 6.0);
	col = mix(col, gold.rgb, 1.0 - smoothstep(petals - 0.6, petals + 0.4, cr));
	col = mix(col, ink.rgb, 1.0 - smoothstep(1.0, 1.8, cr));
	// the vertical axis: a gold stem with lozenges, top and bottom
	for (int i = 0; i < 2; i++) {
		float sy = i == 0 ? -1.0 : 1.0;
		vec2 sp = p - vec2(0.0, sy * 74.0);
		float loz = abs(sp.x) * 1.6 + abs(sp.y) - 8.0;
		col = mix(col, gold.rgb, 1.0 - smoothstep(-0.6, 0.6, loz));
		col = mix(col, ink.rgb, 1.0 - smoothstep(-0.6, 0.6, loz + 4.0));
		float stem = line(p.x, 1.0) * step(44.0, abs(p.y)) * step(abs(p.y), 66.0);
		col = mix(col, gold_dim.rgb, stem);
	}
	COLOR = vec4(col, 1.0);
}
