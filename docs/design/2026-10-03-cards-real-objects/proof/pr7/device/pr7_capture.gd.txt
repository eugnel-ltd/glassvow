var _args: Dictionary = {}
var _main: Main


func _initialize() -> void:
	for arg: String in OS.get_cmdline_user_args():
		var at: int = arg.find("=")
		if at > 0:
			_args[arg.substr(2, at - 2)] = arg.substr(at + 1)
		else:
			_args[arg.trim_prefix("--")] = "true"
	var shape: StringName = StringName(str(_args.get("shape", "pad-landscape")))
	DisplayServer.window_set_size(StageShape.REFERENCES.get(shape, Vector2i(1180, 820)))
	_seed(str(_args.get("mode", "reward")))
	_main = (load("res://application/main.tscn") as PackedScene).instantiate() as Main
	_main._boot_args = PackedStringArray(["--shape=%s" % shape, "--locale=%s" % str(_args.get("locale", "en"))])
	root.add_child(_main)
	_run.call_deferred()


func _seed(mode: String) -> void:
	SaveService.clear_run("", ScenarioKernel.RUN_PATH)
	SaveService.clear_vigil(ScenarioKernel.VIGIL_PATH)
	var vigil: VigilState = VigilState.blank()
	vigil.scenes_seen.append("opening")
	vigil.deeds["wins"] = 1
	SaveService.store_vigil(vigil, ScenarioKernel.VIGIL_PATH)
	var content: ContentDB = ContentDB.load_full()
	var run: RunState = RunState.new_run(content, 65501, "pr7-capture")
	run.map = WorldMap.benchmark(run).to_dict()
	run.act = 1
	run.waystones_lit = 3
	if mode == "reward":
		run.pending_reward = {
			"rewards": {"gold": 24, "cards": ["quakeblow", "resonantLance", "defend"]},
			"taken": {"gold": false, "card": false, "potion": false, "relic": false},
			"slain_enemy": {"id": "duskfang", "hue": 22},
		}
	else:
		run.pending_dawn = {"events": [
			{"kind": "quest", "title": "The Hollow Lamplighter", "body": "—"},
			{"kind": "back", "back": "eclipse", "title": "%TITLE%", "body": "%BODY%"},
		], "cursor": 1}
	SaveService.store(run, ScenarioKernel.RUN_PATH)


func _run() -> void:
	await _frames(4)
	AudioServer.set_bus_mute(0, true)
	var prefs: Preferences = Preferences.new()
	prefs.language = str(_args.get("locale", "en"))
	prefs.diagnostics_notice_seen = true
	prefs.reduce_motion = _args.has("rm")
	Preferences.active = prefs
	_main._title_kindled = true
	_main._show_title()
	await _frames(30)
	var saved: RunState = SaveService.load_run(_main.content, ScenarioKernel.RUN_PATH)
	if str(_args.get("mode", "reward")) == "dawn":
		var dawn: Dictionary = saved.pending_dawn
		var ev: Dictionary = dawn["events"][1]
		ev["title"] = Locale.active.t("ui.back.eclipse.name")
		ev["body"] = Locale.active.t("ui.dawn.backBody")
	_main._hints = null
	_main._continue_run(saved)
	var out: String = str(_args.get("out", "/tmp"))
	var tag: String = "%s-%s%s" % [str(_args.get("shape", "pad-landscape")), str(_args.get("locale", "en")),
		"-rm" if _args.has("rm") else ""]
	if str(_args.get("mode", "reward")) == "reward":
		await _reward(out, tag)
	else:
		await _dawn(out, tag)
	quit()


func _reward(out: String, tag: String) -> void:
	var screen: RewardEmbers = _main._reward_screen
	var t0: float = screen._t
	# Mid-deal, mid-turn (card 1 half way), and at rest.
	for pair: Array in [["a-deal", screen._deal_at() + RewardEmbers.STACK_IN + 0.12],
			["b-turn", screen._turn_at(1) + RewardEmbers.TURN * 0.5],
			["c-rest", screen._entrance_end() + 0.6]]:
		await _until(func() -> bool: return screen._t - t0 >= float(pair[1]) - t0 or screen._settled and pair[0] != "c-rest")
		if pair[0] == "c-rest":
			await _frames(36)
		_shot("%s/reward-%s-%s.jpg" % [out, pair[0], tag])
	# The pick: tap the middle card, then mid-flight and landed.
	var card: CardView = screen._cards[1]
	_tap(card.get_global_rect().get_center())
	await _frames(int(RewardEmbers.PICK_FLY * 60.0 * 0.5))
	_shot("%s/reward-d-fly-%s.jpg" % [out, tag])
	await _frames(int(RewardEmbers.PICK_FLY * 60.0 * 0.5) + 6)
	_shot("%s/reward-e-home-%s.jpg" % [out, tag])
	await _frames(40)
	_shot("%s/reward-f-after-%s.jpg" % [out, tag])
	print("PR7 deck count after pick: ", _main._run_hud._deck_count.text, " deck size ", _main.game.run.player.deck.size())


func _dawn(out: String, tag: String) -> void:
	var screen: DawnScreen = _main._route_screen as DawnScreen
	if screen == null:
		for child: Node in _main.get_children():
			if child is DawnScreen:
				screen = child
	await _frames(int((DawnScreen.BACK_TURN_AT + DawnScreen.BACK_TURN * 0.5) * 60.0))
	_shot("%s/dawn-a-turn-%s.jpg" % [out, tag])
	await _frames(90)
	_shot("%s/dawn-b-rest-%s.jpg" % [out, tag])


func _until(cond: Callable) -> void:
	for _i: int in range(600):
		if cond.call():
			return
		await process_frame


func _shot(file: String) -> void:
	var image: Image = root.get_texture().get_image()
	image.save_jpg(file, 0.9)
	print("PR7 still: ", file)


func _tap(at: Vector2) -> void:
	var move: InputEventMouseMotion = InputEventMouseMotion.new()
	move.position = at
	move.global_position = at
	root.push_input(move)
	for pressed: bool in [true, false]:
		var ev: InputEventMouseButton = InputEventMouseButton.new()
		ev.button_index = MOUSE_BUTTON_LEFT
		ev.pressed = pressed
		ev.position = at
		ev.global_position = at
		root.push_input(ev)


func _frames(count: int) -> void:
	for _i: int in range(count):
		await process_frame
