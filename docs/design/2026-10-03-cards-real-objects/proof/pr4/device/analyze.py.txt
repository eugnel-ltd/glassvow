import sys, glob, os, statistics as st
# usage: analyze.py <batch dir>   (rows_<nonce>.txt from flight_probe.gd)
D = sys.argv[1]
def pct(a, q):
    a = sorted(a)
    if not a: return float("nan")
    k = (len(a) - 1) * q
    f = int(k); c = min(f + 1, len(a) - 1)
    return a[f] + (a[c] - a[f]) * (k - f)
def load(path):
    nonce = os.path.basename(path)[5:-4]
    rows = [l.split() for l in open(path) if l.strip()]
    rows = [r for r in rows if r and r[0] == nonce]
    head = next((" ".join(r[2:]) for r in rows if r[2] == "PROBE"), "")
    build = next((w[6:] for w in head.split() if w.startswith("build=")), "?")
    hand = next((" ".join(r[2:5]) for r in rows if r[2] == "HAND"), "")
    frames = [dict(i=int(r[3]), ph=r[4], wall=float(r[5]), pre=float(r[6]), draw=float(r[7]), vram=float(r[10])) for r in rows if r[2] == "F"]
    return nonce, build, head, hand, frames
runs = []
for path in sorted(glob.glob(os.path.join(D, "rows_*.txt"))):
    nonce, build, head, hand, frames = load(path)
    if not frames: continue
    ph = lambda p: [f for f in frames if f["ph"] == p]
    load_f = ph("load")
    fight = ph("fight")
    deals = {k: ph("deal%d" % k) for k in range(1, 7)}
    measured = [f for k in range(2, 7) for f in deals[k]]
    rest = [f for f in frames if f["ph"].startswith("rest")]
    runs.append(dict(nonce=nonce, build=build, hand=hand, load=load_f, fight=fight, deals=deals, measured=measured, rest=rest, head=head))
def w(fs): return [f["wall"] for f in fs]
print("run   build     hand            | load    N+1    N+2   (draw N+2)  worst N+1..N+10 | deal1 p50/p95/max | deals2-6 n  p50   p95   max  >33 mean | rest p50  p95  mean | deal-rest p50")
for r in runs:
    L = r["load"][0]["wall"] if r["load"] else float("nan")
    f = r["fight"]
    n1 = f[0]["wall"] if len(f) > 0 else float("nan"); n2 = f[1]["wall"] if len(f) > 1 else float("nan")
    d2 = f[1]["draw"] if len(f) > 1 else float("nan")
    worst = max(w(f[:10])) if f else float("nan")
    d1 = w(r["deals"][1]); m = w(r["measured"]); rs = w(r["rest"])
    dl = "%5.1f/%5.1f/%5.1f" % (st.median(d1), pct(d1, .95), max(d1)) if d1 else "   -  "
    if m:
        print("%-5s %-9s %-15s | %6.0f %6.1f %6.1f (%5.1f) %6.1f | %s | %3d %5.2f %5.2f %5.1f %3d %5.2f | %5.2f %5.2f %5.2f | %+5.2f" % (
            r["nonce"], r["build"], r["hand"][:15], L, n1, n2, d2, worst, dl, len(m), st.median(m), pct(m, .95), max(m), sum(1 for x in m if x > 33.4), st.mean(m),
            st.median(rs), pct(rs, .95), st.mean(rs), st.median(m) - st.median(rs)))
    else:
        print("%-5s %-9s %-15s | %6.0f %6.1f %6.1f (%5.1f) %6.1f | %s | (no measured deals)" % (r["nonce"], r["build"], r["hand"][:15], L, n1, n2, d2, worst, dl))
print()
by = {}
for r in runs:
    if r["measured"] and not r["nonce"].startswith(("w", "v", "s")):
        by.setdefault(r["build"], []).append(r)
for b, rs in by.items():
    meds = [st.median(w(r["measured"])) for r in rs]; p95s = [pct(w(r["measured"]), .95) for r in rs]
    rmeds = [st.median(w(r["rest"])) for r in rs]; rp95 = [pct(w(r["rest"]), .95) for r in rs]
    pooled = [x for r in rs for x in w(r["measured"])]; prest = [x for r in rs for x in w(r["rest"])]
    print("%s: %d launches | deal per-launch p50s %s -> batch median %.2f | p95s %s -> batch median %.2f | pooled p50 %.2f p95 %.2f >33: %d of %d | rest batch median p50 %.2f p95 %.2f (pooled %.2f / %.2f)" % (
        b, len(rs), [round(x, 2) for x in meds], st.median(meds), [round(x, 2) for x in p95s], st.median(p95s),
        st.median(pooled), pct(pooled, .95), sum(1 for x in pooled if x > 33.4), len(pooled), st.median(rmeds), st.median(rp95), st.median(prest), pct(prest, .95)))
print()
print("deal frames over 33.4 ms (run, deal, frame in deal, wall, to-draw, draw):")
for r in runs:
    for k in range(1, 7):
        fs = r["deals"][k]
        for j, f in enumerate(fs):
            if f["wall"] > 33.4:
                print("  %s %s deal%d #%d wall %.1f to_draw %.1f draw %.1f" % (r["nonce"], r["build"], k, j, f["wall"], f["pre"], f["draw"]))
