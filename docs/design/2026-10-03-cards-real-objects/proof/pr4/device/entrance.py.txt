import sys, glob, os, statistics as st
# usage: entrance.py <batch dir> [first-k]   the load frame and the entrance frames per run,
# each as wall(to-draw+draw), long frames (>= 33.4 ms) starred.
D = sys.argv[1]; K = int(sys.argv[2]) if len(sys.argv) > 2 else 16
def pct(a, q):
    a = sorted(a); k = (len(a) - 1) * q; f = int(k); c = min(f + 1, len(a) - 1)
    return a[f] + (a[c] - a[f]) * (k - f)
fixed = {}
for path in sorted(glob.glob(os.path.join(D, "rows_*.txt"))):
    nonce = os.path.basename(path)[5:-4]
    rows = [l.split() for l in open(path) if l.strip()]
    rows = [r for r in rows if r and r[0] == nonce]
    head = next((" ".join(r[2:]) for r in rows if r[2] == "PROBE"), "")
    build = next((w[6:] for w in head.split() if w.startswith("build=")), "?")
    F = [(r[4], float(r[5]), float(r[6]), float(r[7])) for r in rows if r[2] == "F"]
    ent = [f for f in F if f[0] in ("load", "fight")][:K]
    s = " ".join(("%d:%s%.0f(%.0f+%.0f)" % (i, "*" if f[1] >= 33.4 else "", f[1], f[2], f[3])) for i, f in enumerate(ent))
    longs = [i for i, f in enumerate([f for f in F if f[0] == "fight"][:60]) if f[1] >= 33.4]
    print("%-5s %-9s %s | long fight frames (N+i, first 60): %s" % (nonce, build, s, [i + 1 for i in longs]))
    # The first 48 frames (0.8 s) of each measured deal: one window for both builds.
    if not nonce.startswith(("w", "v", "s")):
        for k in range(2, 7):
            d = [f[1] for f in F if f[0] == "deal%d" % k][:48]
            fixed.setdefault(build, []).extend(d)
print()
for b, xs in [(b, xs) for b, xs in fixed.items() if xs]:
    print("%s fixed 48-frame deal windows (deals 2-6): n=%d p50 %.2f p95 %.2f mean %.3f max %.1f" % (b, len(xs), st.median(xs), pct(xs, .95), st.mean(xs), max(xs)))
